
FormalSystem Predicate:

    # Inherit from the Propositional logic system
    inherit propositional
    
    # Variables, constants, functions, and predicates are all abstract patterns
    Abstract variable, constant, function, predicate
    
    UnionPattern term:
        variable
        constant

    UnionPattern comma_separated_terms:
        with t as term, cst as comma_separated_terms:
            t
            cst, t

    comma_separated_terms.terms():
        instances(term)

    term:
        with $f$ as function, $T$ as comma_separated_terms:
            $f(T)$

    UnionPattern atomic_formula:
        with $t_1$, $t_2$ as term:
            $t_1 = t_2$
            
        with $P$ as predicate, $T$ as comma_separated_terms:
            $P(T)$

    Pattern forall:
        with $x$ as variable, $\alpha$ as formula:
            $\forall x \; \alpha$
            
    # Add atomic_formula and forall to formula union pattern
    formula:
        atomic_formula
        forall
        
    variable.free():
        # Check if the variable is free in the whole tree
        return not has_parent(f as forall; f.x == self)
        
    variable.free_in(alpha as formula):
        # Check if the variable is free in the given formula.
        return not has_parent(f as forall; f.x == self and f.is_descendant_of(alpha))

    formula.free_variables():
        return instances(v as variable; v.free_in(self))

    formula.bound_variables():
        return instances(v as variable; (not v.free_in(self)) and (not v.parent().pattern() == forall))

    term.is_free_for(x as variable, alpha as formula):
        return alpha.free_variables().each(instance):
            (not instance == x) or (not instance.has_parent(f as forall; f.x.equal_any(self.instances(variable))))

    # Add conditions to proof context
    ProofContext:
        conditions: set()
        
    # Overwrite the logical maps_to function to include conditions
    logical.maps_to(other as logical):
    
        # Start with mapping given in the reference
        mapping = other.reference_mapping()
        
        # Get the formula mapping
        result = formula().maps_to(other.formula(), mapping)
        if result is False:
            return False
            
        # Get the given mapping
        result = given.maps_to(other.given, mapping)
        if result is False:
            return False
            
        # Check all conditions are met or mapped
        for c in self.conditions:
            if other.check_condition(c, mapping) or c.maps_into_set(other.conditions, mapping):
                continue
            
            # Condition fails
            return False
        
        # Otherwise ok
        return True
        
    # Allow system conditions in if
    Pattern if_condition_pattern:
        # _system_condition_ is not a pattern but matches only if the given
        # string parses successfully as a Condition.
        with sc as _system_condition_:
            if sc:
            
    LineType if_condition:
        pattern: if_condition_pattern
        behaviour: indent
        context.conditions:
            # Add condition to context
            union: match().sc.condition().conjunctive_parts()
        
    # Overwrite the inference rules to include context conditions
    with $\alpha$, $\beta$ as formula:
        InferenceRule Modus_Ponens:
            label:
                MP

            antecedents:
                $\alpha$
                $(\alpha \rightarrow \beta)$
            deduction:
                $\beta$

            condition:
                # Conditional on the deduction given set containing the antecedent given sets
                (antecedents[0].given.union(antecedents[1].given).issubset(deduction.given)) and (antecedents[0].conditions.union(antecedents[1].conditions).issubset(deduction.conditions))

        InferenceRule Deduction_Theorem_1:
            label:
                DT1

            antecedents:
                $(\alpha \rightarrow \beta)$
                
            deduction:
                $\beta$

            condition:
                (antecedent.given.issubset(deduction.given)) and (alpha in deduction.given) and (antecedent.conditions.issubset(deduction.conditions))

        InferenceRule Deduction_Theorem_2:
            label:
                DT2

            antecedents:
                $\beta$

            deduction:
                $(\alpha \rightarrow \beta)$

            condition:
                (deduction.given.union(set(alpha)) == antecedent.given) and (antecedent.conditions == deduction.conditions)

        InferenceRule Assumed:
            label:
                As
                
            antecedents:
                # No antecedents!
                
            deduction:
                $\alpha$
                
            condition:
                alpha in deduction.given
                
        InferenceRule Thinning:
            label:
                T

            antecedents:
                $\alpha$

            deduction:
                $\alpha$

            condition:
                (antecedent.given.issubset(deduction.given)) and (antecedent.conditions.issubset(deduction.conditions))

        InferenceRule Theorem:
            label:
                Th

            antecedents:
                $\alpha$

            deduction:
                $\beta$

            condition:
                antecedent.maps_to(deduction)
                
        InferenceRule Definition:
            label:
                Def
                
            antecedents:
                labelled_definition_pattern
                $\alpha$
                
            deduction:
                $\beta$
                
            condition:
                (deduction.follows_from_definition(antecedents[1], antecedents[0].definition())) and (antecedents[1].given.issubset(deduction.given)) and (antecedents[1].conditions.issubset(deduction.conditions))
