
# Create a format dictionary for tex
Format tex:
    # Treat multiple spaces between characters as a single space
    (?P<char1>[^\s])\s\s+(?P<char2>[^\s]): \g<char1> \g<char2>
    
    # Ignore $...$ surrounding formulae
    \$(?P<inner>.*?)\$: \g<inner>
    
    # Ignore \; spacing
    (?P<char>[^;]*);: \g<char>
    
    # Ignore \ characters
    (?P<char>[^\\]*)\\: \g<char>
    
    
FormalSystem Predicate:

    # Use tex formatting throughout
    Format tex
    
    # Variables and constants are abstract patterns
    Abstract variable, constant

    Regex integer:
        ^(?:0|[1-9][0-9]*)$

    Regex variable_name:
        ^[a-zA-Z][a-zA-Z0-9_]*$

    UnionPattern comma_separated_variable_names:
        with var as variable_name, csv as comma_separated_variable_names:
            var
            csv, var

    UnionPattern term:
        variable
        constant

    UnionPattern comma_separated_terms:
        with t as term, cst as comma_separated_terms:
            t
            cst, t

    comma_separated_terms.terms():
        instances(term)

    term:
        with $i$ as integer, $T$ as comma_separated_terms:
            $f_i(T)$

    UnionPattern atomic_formula:
        with $t_1$, $t_2$ as term:
            $t_1 = t_2$
            
        with $i$ as integer, $T$ as comma_separated_terms:
            $P_i(T)$

    Pattern rightarrow:
        $(\alpha \rightarrow \beta)$

    Pattern not:
        $\neg \alpha$

    Pattern forall:
        with $x$ as variable:
            $\forall x \; \alpha$

    UnionPattern formula:
        atomic_formula
        rightarrow
        not
        forall

    # Add formula as variable in recursive patterns
    rightarrow.add_variables:
        $\alpha$: formula
        $\beta$: formula

    not.add_variables:
        $\alpha$: formula

    forall.add_variables:
        $\alpha$: formula

    UnionPattern comma_separated_formulae:
        with f as formula, csf as comma_separated_formulae:
            f
            csf, f

    variable.free():
        return not has_parent(f as forall; f.x == self)

    formula.free_variables():
        return instances(v as variable; v.free())

    formula.bound_variables():
        return instances(v as variable; (not v.free()) and (not v.parent().pattern() == forall))

    term.is_free_for(x as variable, alpha as formula):
        return alpha.free_variables().each(instance):
            (not instance == x) or (not instance.has_parent(f as forall; f.x.equal_any(self.instances(variable))))

    # Create a default proof context
    ProofContext:
        given: MatchSet()
        conditions: set()
        
    Regex reference:
        ^[a-zA-Z0-9_,\. ]+$

    Pattern without_label:
        with fo as formula, refs as reference:
            fo ref{refs}
            
    Pattern with_label:
        with fo as formula, refs as reference, label_ref as reference:
            fo ref{refs} label{label_ref}
            
    Pattern axiom_pattern:
        with fo as formula, label_ref as reference:
            fo label{Ax: label_ref}
            
    UnionPattern logical_pattern:
        without_label
        with_label
        axiom_pattern
            
    # Clear tex format from logical_pattern and it's options - to keep tex 
    # formatting which is used for displaying the formula
    logical_pattern.format:
        clear
        
    without_label.format:
        clear
        
    with_label.format:
        clear
        
    axiom_pattern.format:
        clear
        
    logical_pattern.formula():
        # Get the formula regardless of the option
        return union_submatch().fo
        
    logical_pattern.reference():
        # Get any reference
        return union_submatch().refs.string()
        
    logical_pattern.label():
        # Get any label
        return union_submatch().label_ref.string()
        
    logical_pattern.display():
        # The display value - ignore references and labels
        return union_submatch().fo.string()
        
    logical_pattern.axiom():
        # The axiom formula - if it exists
        return axiom_pattern.fo.string()

    LineType logical:
        pattern: logical_pattern
        behaviour: logical
        
    # Function to get formula map
    logical.formula_mapping(other as logical):
        # Returns a mapping or False
        
        mapping = dict()
        result = formula().maps_to(other.formula(), mapping)
        
        if mapping is False:
            return False
        
        return mapping
        
    # Function to get map for given set as well as formula
    logical.given_mapping(other as logical):
        # Returns a mapping or False
        
        # Get the formula mapping
        mapping = self.formula_mapping(other)
        
        if mapping is False:
            return False
            
        if given.maps_to(other.given, mapping):
            return mapping
        
        return False
        
    # Check conditions from another line are consistent with this one
    logical.check_consistent_conditions(other as logical, mapping as dict):
        # Returns a boolean. Mapping is updated within maps_into_set as needed.
        for c in self.conditions:
            if other.check_condition(c, mapping) or c.maps_into_set(other.conditions, mapping):
                continue
            
            # Condition fails
            return False
            
        return True
    
    # Function to check the proof line follows from other.
    logical.maps_to(other as logical):
    
        # Get the given mapping
        mapping = self.given_mapping(other)
        
        if mapping is False:
            return False
    
        return self.check_consistent_conditions(other, mapping)

    Pattern let_pattern:
        with csv as comma_separated_variable_names, p as variable_name:
            let csv be p
           
    let_pattern.keys():
        return csv.instances(variable_name).strings()

    LineType let:
        pattern: let_pattern
        behaviour: none
        context.string_variables:
            # Extend with all the keys
            match().keys(): match().p.lookup()

    Regex comment_pattern:
        ^#.*$

    LineType comment:
        pattern: comment_pattern
        behaviour: none

    Pattern if_pattern:
        with csf as comma_separated_formulae:
            if csf:
           
    if_pattern.parts():
        return csf.shallow_instances(formula)

    LineType if:
        pattern: if_pattern
        behaviour: indent
        context.given:
            # Add pattern parts to given set
            union: match().parts()
            
    Pattern if_condition_pattern:
        # _system_condition_ is not a pattern but matches only if the given
        # string parses successfully as a Condition.
        with sc as _system_condition_:
            if sc:
            
    LineType if_condition:
        pattern: if_condition_pattern
        behaviour: indent
        context.conditions:
            # Add condition to context
            add: match().sc.condition()
            
    Regex import_path:
        ^[a-zA-Z0-9_\.]+$
        
    Pattern import_pattern:
        with path as import_path, label as variable_name:
            import path as label
            
    import_pattern.path():
        return path.string()
        
    import_pattern.label():
        return label.string()
            
    LineType import:
        pattern: import_pattern
        behaviour: import
        
    Regex any:
        ^.+$
        
    Pattern definition_pattern:
        with lower as any, higher as any, pattern as variable_name:
            define higher as lower for pattern
            
    definition_pattern.format:
        clear
            
    Pattern labelled_definition_pattern:
        with def as definition_pattern, label_ref as reference:
            def label{label_ref}
            
    labelled_definition_pattern.format:
        clear
            
    labelled_definition_pattern.display():
        return def.string()
        
    labelled_definition_pattern.label():
        return label_ref.string()
        
    labelled_definition_pattern.higher():
        return def.higher
        
    labelled_definition_pattern.lower():
        return def.lower
        
    labelled_definition_pattern.for():
        return def.pattern.lookup()
            
    LineType definition:
        pattern: labelled_definition_pattern
        behaviour: definition
        
    with $\alpha$, $\beta$ as formula:
        InferenceRule Modus_Ponens:
            label:
                MP

            antecedents:
                $\alpha$
                $(\alpha \rightarrow \beta)$
            deduction:
                $\beta$

            condition:
                # Conditional on the deduction given set containing the antecedent given sets
                (antecedents[0].given.union(antecedents[1].given).issubset(deduction.given)) and (antecedents[0].conditions.union(antecedents[1].conditions).issubset(deduction.conditions))

        InferenceRule Deduction_Theorem_1:
            label:
                DT1

            antecedents:
                $(\alpha \rightarrow \beta)$
                
            deduction:
                $\beta$

            condition:
                (antecedent.given.issubset(deduction.given)) and (alpha in deduction.given) and (antecedent.conditions.issubset(deduction.conditions))

        InferenceRule Deduction_Theorem_2:
            label:
                DT2

            antecedents:
                $\beta$

            deduction:
                $(\alpha \rightarrow \beta)$

            condition:
                (deduction.given.union(set(alpha)) == antecedent.given) and (antecedent.conditions == deduction.conditions)

        InferenceRule Assumed:
            label:
                As
                
            antecedents:
                # No antecedents!
                
            deduction:
                $\alpha$
                
            condition:
                alpha in deduction.given
                
        InferenceRule Thinning:
            label:
                T

            antecedents:
                $\alpha$

            deduction:
                $\alpha$

            condition:
                (antecedent.given.issubset(deduction.given)) and (antecedent.conditions.issubset(deduction.conditions))

        InferenceRule Theorem:
            label:
                Th

            antecedents:
                $\alpha$

            deduction:
                $\beta$

            condition:
                antecedent.maps_to(deduction)
                
        InferenceRule Definition:
            label:
                Def
                
            antecedents:
                labelled_definition_pattern
                $\alpha$
                
            deduction:
                $\beta$
                
            condition:
                (deduction.follows_from_definition(antecedents[1], antecedents[0].definition())) and (antecedents[1].given.issubset(deduction.given)) and (antecedents[1].conditions.issubset(deduction.conditions))
